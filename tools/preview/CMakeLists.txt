cmake_minimum_required(VERSION 3.20)
project(passport_preview C)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(LVGL_DIR "${ROOT}/managed_components/lvgl__lvgl" CACHE PATH "Pinned LVGL 9.5.0")
set(PASSPORT_ASSETS "${ROOT}/assets" CACHE PATH "Generated assets")
set(PASSPORT_MOTION "${PASSPORT_ASSETS}/images/passport_motion.c")
if(NOT EXISTS "${PASSPORT_MOTION}")
    set(PASSPORT_MOTION "${ROOT}/assets/images/passport_motion.c")
endif()
set(PASSPORT_AGENT_AVATAR "${PASSPORT_ASSETS}/images/passport_agent_avatar.c")
set(PASSPORT_BOLD_FONT "${ROOT}/assets/fonts/passport_font_24_bold.c")
if(EXISTS "${PASSPORT_ASSETS}/fonts/passport_font_24_bold.c")
    set(PASSPORT_BOLD_FONT "${PASSPORT_ASSETS}/fonts/passport_font_24_bold.c")
endif()
if(NOT EXISTS "${PASSPORT_AGENT_AVATAR}")
    set(PASSPORT_AGENT_AVATAR "${ROOT}/assets/images/passport_agent_avatar.c")
endif()
file(GLOB_RECURSE LVGL_SOURCES CONFIGURE_DEPENDS "${LVGL_DIR}/src/*.c")
add_library(lvgl STATIC ${LVGL_SOURCES})
target_include_directories(lvgl PUBLIC "${LVGL_DIR}")
target_compile_definitions(lvgl PUBLIC LV_CONF_SKIP=1 LV_COLOR_DEPTH=16 LV_MEM_SIZE=65536 LV_USE_QRCODE=1 LV_USE_LOG=0 LV_FONT_MONTSERRAT_14=1 LV_FONT_MONTSERRAT_20=1 LV_FONT_MONTSERRAT_28=1)
add_executable(passport_preview "${ROOT}/tests/test_passport_ui.c" "${ROOT}/main/passport_ui.c" "${ROOT}/main/passport_model.c" "${ROOT}/main/passport_game.c" "${ROOT}/main/passport_profile.c" "${ROOT}/main/passport_qr.c" "${ROOT}/main/passport_avatar_host.c" "${PASSPORT_ASSETS}/fonts/passport_font_16.c" "${PASSPORT_ASSETS}/fonts/passport_font_20.c" "${PASSPORT_ASSETS}/images/passport_avatar.c" "${PASSPORT_ASSETS}/passport/passport_glyphs.c")
target_include_directories(passport_preview PRIVATE "${ROOT}/main" "${PASSPORT_ASSETS}/passport" "${LVGL_DIR}/src/libs/qrcode")
target_link_libraries(passport_preview PRIVATE lvgl m)
target_sources(passport_preview PRIVATE "${ROOT}/main/passport_animation.c" "${ROOT}/main/passport_scene.c" "${PASSPORT_MOTION}" "${PASSPORT_AGENT_AVATAR}" "${PASSPORT_BOLD_FONT}")
target_compile_options(passport_preview PRIVATE -Wall -Wextra -Werror)
target_compile_definitions(lvgl PUBLIC LV_FONT_FMT_TXT_LARGE=1)
target_link_options(passport_preview PRIVATE -Wl,--wrap=lv_obj_create)
target_link_options(passport_preview PRIVATE -Wl,--wrap=passport_image_current)
target_link_options(passport_preview PRIVATE -Wl,--wrap=passport_avatar_current)
